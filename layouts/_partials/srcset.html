{{/* Image sizes for one resource: 640, 960, 1280, 1920 and 2560 where smaller than the source, plus the source width itself (capped at 2560). 2560 keeps full-width images sharp on retina screens; 960 narrows the jumps for mid-size layouts. Returns dict: src (largest), srcset, width, height. Shared by img.html and posts.json. */}} {{ $widths := slice }} {{ range slice 640 960 1280 1920 2560 }}{{ if lt . $.Width }}{{ $widths = $widths | append . }}{{ end }}{{ end }} {{ $widths = $widths | append (math.Min $.Width 2560 | int) }} {{ $set := slice }} {{ $img := . }} {{ range $widths }} {{ $img = $.Resize (printf "%dx webp" .) }} {{ $set = $set | append (printf "%s %dw" $img.RelPermalink .) }} {{ end }} {{ return dict "src" $img.RelPermalink "srcset" (cond (gt (len $set) 1) (delimit $set ", ") "") "width" $img.Width "height" $img.Height }}
